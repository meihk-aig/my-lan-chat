    import const_val as c
    import socket as sk
    import threading as thd


    class ChatManager:
        def __init__(self, tcp_port, host):
            self.max_length = 1024 * 1024

            self.tcp_port = tcp_port
            self.host = host
            self.connections = {}
            self.active_chat = None

            self.lock = thd.Lock()

            self.stop_event = thd.Event()

            self.listen_socket = sk.socket(sk.AF_INET, sk.SOCK_STREAM)
            self.listen_socket.setsockopt(sk.SOL_SOCKET, sk.SO_REUSEADDR, True)

            self.listen_socket.bind(("", tcp_port))
            self.listen_socket.listen()
            self.listen_socket.settimeout(2)

            self.listener_thread = thd.Thread(target=self.accept, daemon=True)
            self.listener_thread.start()

        def accept(self):
            while not self.stop_event.is_set():
                try:
                    conn, addr = self.listen_socket.accept()
                except sk.timeout:
                    continue

                with self.host.peers_lock:
                    peers = [
                        name
                        for name, info in self.host.peers.items()
                        if info.get("ip") == addr[0]
                    ]

                if len(peers) != 1:
                    conn.close()
                    continue

                self.register_connection(peers[0], conn)

        def register_connection(self, peer_name, conn):
            with self.lock:
                if peer_name in self.connections:
                    conn.close()
                    return

                self.connections[peer_name] = conn

            recv_thread = thd.Thread(
                target=self.recieve_loop, args=(peer_name, conn), daemon=True
            )
            recv_thread.start()

        def connect_to_peer(self, peer_name):
            with self.lock:
                if peer_name in self.connections:
                    return True

            with self.host.peers_lock:
                info = self.host.peers.get(peer_name)

            if info is None:
                print(f"[*] {peer_name} not found.")
                return False

            conn = sk.socket(sk.AF_INET, sk.SOCK_STREAM)
            conn.settimeout(5)

            addr = (info["ip"], info["port"])
            try:
                conn.connect(addr)
            except:
                conn.close()
                return False

            conn.settimeout(None)
            self.register_connection(peer_name, conn)

            with self.lock:
                return self.connections.get(peer_name) is conn

        def close_connection(self, peer_name):
            with self.lock:
                conn = self.connections.pop(peer_name, None)

            if conn:
                conn.close()

        def start_chat(self, peer_name):
            if not self.connect_to_peer(peer_name):
                print(f"[*] Cannot connect to {peer_name}")
                return False

            with self.lock:
                self.active_chat = peer_name

            with self.host.unread_lock:
                messages = self.host.unread.pop(peer_name, [])

                for message in messages:
                    print(f"[{peer_name}] {message}")

            return True

        def leave_chat(self):
            with self.lock:
                self.active_chat = None

        def recv_exact(self, conn, size):
            data = bytearray()

            while len(data) < size:
                chunk = conn.recv(size - len(data))

                if not chunk:
                    return None
                data.extend(chunk)

            return bytes(data)

        def recv_message(self, conn):
            header = self.recv_exact(conn, c.HEADER)
            if header is None:
                return None

            length = int(header.decode(c.FORMAT).strip())

            if not (0 <= length <= self.max_length):
                return None

            payload = self.recv_exact(conn, length)
            if payload is None:
                return None

            return payload.decode(c.FORMAT)

        def send_message(self, peer_name, message):
            with self.lock:
                conn = self.connections.get(peer_name)

            if conn is None:
                return False

            try:
                payload = message.encode(c.FORMAT)
                header = str(len(payload)).ljust(c.HEADER).encode(c.FORMAT)

                conn.sendall(header + payload)

                return True
            except:
                self.close_connection(peer_name)
                return False

        def recieve_loop(self, peer_name, conn):
            try:
                while not self.stop_event.is_set():
                    message = self.recv_message(conn)

                    if message is None:
                        break

                    with self.lock:
                        active = self.active_chat == peer_name

                        if not active:
                            with self.host.unread_lock:
                                unread = self.host.unread.setdefault(peer_name, [])
                                unread.append(message)
                                count = len(unread)

                    if active:
                        print(f"[{peer_name}] {message}")
                    else:
                        print(f"[*] New message from {peer_name} ({count} unread)")
            except:
                with self.lock:
                    if self.connections.get(peer_name) is conn:
                        self.connections.pop(peer_name)

                conn.close()

    import const_val as c
    import peer as pr
    import threading as thd
    import time as tm
    import socket as sk


    class Broadcaster:
        def __init__(self, host, bcast, tcp_port):
            self.recieve_socket = self.make_udp_socket()
            self.broadcast_socket = self.make_udp_socket()

            self.recieve_thread = thd.Thread(target=self.recieve_message, args=(host,))
            self.recieve_thread.daemon = True

            self.broadcast_thread = thd.Thread(
                target=self.broadcast_message, args=(host, bcast, tcp_port)
            )
            self.broadcast_thread.daemon = True

            self.stop_flag = thd.Event()

        def make_udp_socket(self):
            # Create an UDP socket with SOCK_DGRAM parameter
            conn = sk.socket(sk.AF_INET, sk.SOCK_DGRAM)
            # The parameters SO_REUSEADDR and True mean that the address (IP and port) can be used by others
            conn.setsockopt(sk.SOL_SOCKET, sk.SO_REUSEADDR, True)
            # The parameters SO_BROADCAST and True make sure that this socket cannot be denied by the operating system
            conn.setsockopt(sk.SOL_SOCKET, sk.SO_BROADCAST, True)
            # Bind the socket to the address as below ("" means all the machine in the LAN)
            conn.bind(("", c.PORT))

            return conn

        def recieve_message(self, host):
            print(f"[*] Broadcasting presence on UDP port {c.PORT}")
            while True:
                # The variable data holds the message in bytes (we have to decode it), and addr holds the sender IP and PORT
                data, addr = self.recieve_socket.recvfrom(c.HEADER)
                # Decode the message
                msg = data.decode(c.FORMAT, errors="replace")
                # Split the message into parts to get the information
                info = msg.split("|")
                # Skip the broadcast of this socket
                if len(info) == 4 and info[0] == "!LOL" and info[1] == host.name:
                    continue

                peer_name = info[1]
                tcp_port = int(info[2])
                peer_ip = addr[0]
                peer_state = info[3]

                # Add the information recieved to the peers dictionary which is a global variable
                with host.peers_lock:
                    host.peers[peer_name] = {
                        "ip": peer_ip,
                        "port": tcp_port,
                        "last_seen": tm.time(),
                        "state": peer_state,
                    }

                if self.stop_flag.is_set():
                    break

        def broadcast_message(self, host, bcast, tcp_port):
            print(f"[*] Listening on TCP port {tcp_port}")
            while True:
                # Make the message and encode it to bytes
                msg = f"!LOL|{host.name}|{tcp_port}|{host.state}".encode(c.FORMAT)
                # The variable addr holds IP limits (defalt = 255.255.255.255) and the PORT that the others are listening to
                addr = (bcast, c.PORT)
                # Send the message and the adress
                self.broadcast_socket.sendto(msg, addr)
                # If the stop event happens, these two lines will terminate the process
                if self.stop_flag.is_set():
                    break

                # Stop 3 second before sending another message
                tm.sleep(3)

        def start_listening(self):
            self.recieve_thread.start()

        def start_broadcasting(self):
            self.broadcast_thread.start()

        def end_process(self):
            self.stop_flag.set()

    import threading as thd


    class Peer:
        def __init__(self, peer_name):
            self.name = peer_name
            self.state = "online"

            self.peers = {}
            self.peers_lock = thd.Lock()

            self.unread = {}
            self.unread_lock = thd.Lock()


    def print_help_message():
        print("[*] Available commands: ")
        print(
            "    /list                     Show online users with number, name, IP and status"
        )
        print(
            "    /chat <name|number>       Open a chat with the selected user (TCP connection)"
        )
        print("    <any text>                Send a message to the current chat partner")
        print("    /send <file>              Send a file to the current chat partner")
        print("    /leave                    Close the current chat")
        print("    /status <online|away>     Change your status (sent via UDP)")
        print("    /help                     Show all commands")
        print("    /quit                     Exit the program")

    from broadcast import *
    import peer as p
    import const_val as c
    import chatting as cht

    import threading as thd
    import time as tm
    import socket as sk
    import argparse as ap

    if __name__ == "__main__":
        try:
            # These are parser from the argparse module
            # Eg: Run "python .\src\main.py --name Mathhew" to start the program and set your name Matthew

            parser = ap.ArgumentParser()
            parser.add_argument("--name", required=True)
            parser.add_argument("--bcast", default="255.255.255.255")
            parser.add_argument("--tcp-port", type=int, default=5050)

            lst = parser.parse_args()

            host = p.Peer(lst.name)

            broadcaster = Broadcaster(host, lst.bcast, lst.tcp_port)
            broadcaster.start_broadcasting()
            broadcaster.start_listening()

            print("Hello")

            manager = cht.ChatManager(lst.tcp_port, host)

            print("Hello")

            print("[*] Type /help for instructions")
            while True:
                command = input("> ").split()

                if len(command) == 0:
                    continue

                if command[0] == "/quit":  # /quit
                    broadcaster.end_process()
                    print("[*] Connection closed")
                    break

                elif command[0] == "/help":  # /help
                    p.print_help_message()

                elif command[0] == "/list":  # /list
                    current_time = tm.time()

                    with host.peers_lock:
                        index = 1
                        peers_snapshot = host.peers.copy()
                        for name, info in peers_snapshot.items():
                            current_state = info["state"]

                            if current_time - info["last_seen"] > c.TIMEOUT:
                                current_state = "away"

                            print(f"    {index}. {name} ({info['ip']}) {current_state}")

                            index += 1

                elif command[0] == "/status" and len(command) == 2:  # /status
                    if command[1] == "online":
                        host.state = "online"
                    elif command[1] == "away":
                        host.state = "away"
                    else:
                        print('[*] Your status must be "online" or "away"')
                        continue

                    print(f'[*] Your status set to "{host.state}"')

                elif command[0] == "/chat":
                    if len(command) != 2:
                        print("[*] /chat syntax must be /chat <peer_name>")
                        continue

                    peer_name = command[1]

                    manager.start_chat(peer_name)

                    while True:
                        message = input("> ").strip()

                        if not message:
                            continue

                        if message == "/leave":
                            print(f"[*] You have left the chat with {peer_name}")
                            manager.leave_chat()
                            break

                        manager.send_message(peer_name, message)

        except KeyboardInterrupt:
            print("[*] Connection closed")
